```python
# strategies/xau_arbitrage/quotes.py

async def _cycle_once(self) -> None:
    # 🔧 在下单前检查保证金
    balance = await self._check_margin()
    if not balance['sufficient']:
        logger.warning(
            f"Insufficient margin: available={balance['available']}, "
            f"required={balance['required']}, pausing quotes..."
        )
        return  # 跳过本次做市
    
    # ... 继续下单 ...

async def _check_margin(self) -> dict:
    """检查保证金是否充足"""
    try:
        balance = await asyncio.to_thread(self.exchange.client.fetch_balance)
        
        # 计算需要的保证金
        total_usd = sum(self.order_sizes) * 2  # 买 + 卖
        price = self.feed.get_bbo("aster").mid
        contracts = total_usd / price
        
        # 假设 5x 杠杆
        required_margin = (contracts * price) / 5
        
        # 获取可用保证金
        available = balance['USDT']['free'] + balance.get('USDF', {}).get('free', 0)
        
        return {
            'sufficient': available >= required_margin * 1.2,  # 留 20% 缓冲
            'available': available,
            'required': required_margin,
        }
    except Exception as e:
        logger.error(f"Failed to check margin: {e}")
        return {'sufficient': False, 'available': 0, 'required': 0}
```

## 🚨 紧急情况处理

### 如果已经产生单边敞口怎么办？

```python
# 新建 scripts/emergency_hedge.py

"""紧急对冲工具：手动对冲单边仓位"""

import asyncio
from api import get_client
from dotenv import load_dotenv
import os

async def emergency_hedge():
    load_dotenv()
    
    # 初始化交易所
    aster = get_client('aster', {
        'apiKey': os.getenv('ASTER_API_KEY'),
        'secret': os.getenv('ASTER_SECRET_KEY'),
    })
    
    backpack = get_client('backpack', {
        'apiKey': os.getenv('BACKPACK_KEY'),
        'secret': os.getenv('BACKPACK_SECRET'),
    })
    
    # 获取仓位
    aster_pos = aster.fetch_positions('XAU/USDT')
    backpack_pos = backpack.fetch_positions('PAXG/USDC:USDC')
    
    print(f"Aster position: {aster_pos}")
    print(f"Backpack position: {backpack_pos}")
    
    # 计算不平衡
    aster_contracts = float(aster_pos[0]['contracts']) if aster_pos else 0
    backpack_contracts = float(backpack_pos[0]['contracts']) if backpack_pos else 0
    
    imbalance = aster_contracts + backpack_contracts
    
    print(f"\nImbalance: {imbalance:.4f}")
    
    if abs(imbalance) < 0.0001:
        print("✅ Positions are balanced!")
        return
    
    # 确认对冲
    print(f"\n⚠️ Need to hedge {abs(imbalance):.4f} contracts")
    confirm = input("Proceed with emergency hedge? (yes/no): ")
    
    if confirm.lower() != 'yes':
        print("Cancelled.")
        return
    
    # 执行对冲
    if imbalance > 0:
        # 需要在 Backpack 上卖出
        print(f"Selling {imbalance:.4f} PAXG on Backpack...")
        order = backpack.create_market_order('PAXG/USDC:USDC', 'sell', imbalance)
        print(f"✅ Order placed: {order}")
    else:
        # 需要在 Backpack 上买入
        print(f"Buying {abs(imbalance):.4f} PAXG on Backpack...")
        order = backpack.create_market_order('PAXG/USDC:USDC', 'buy', abs(imbalance))
        print(f"✅ Order placed: {order}")

if __name__ == '__main__':
    asyncio.run(emergency_hedge())
```
